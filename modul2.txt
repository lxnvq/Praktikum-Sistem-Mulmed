<script>

    // Geolocation API
    function getLocation() {

        const result = document.getElementById("locationResult");

        if (navigator.geolocation) {

            navigator.geolocation.getCurrentPosition(
                showPosition,
                showError
            );

        } else {
            result.innerHTML =
                "Geolocation tidak didukung oleh browser.";
        }
    }


    function showPosition(position) {

        document.getElementById("locationResult").innerHTML =
            "Latitude: " + position.coords.latitude +
            "<br>Longitude: " + position.coords.longitude +
            "<br>Akurasi: " + position.coords.accuracy + " meter";
    }


    function showError(error) {

        let message = "";

        switch (error.code) {

            case error.PERMISSION_DENIED:
                message = "Pengguna menolak permintaan lokasi.";
                break;

            case error.POSITION_UNAVAILABLE:
                message = "Informasi lokasi tidak tersedia.";
                break;

            case error.TIMEOUT:
                message = "Permintaan lokasi terlalu lama.";
                break;

            default:
                message = "Terjadi kesalahan.";
        }

        document.getElementById("locationResult").innerHTML = message;
    }


    // Drag and Drop API
    function allowDrop(event) {
        event.preventDefault();
    }


    function drag(event) {
        event.dataTransfer.setData("text", event.target.id);
    }


    function drop(event) {

        event.preventDefault();

        const data = event.dataTransfer.getData("text");

        event.target.appendChild(
            document.getElementById(data)
        );
    }

</script>
